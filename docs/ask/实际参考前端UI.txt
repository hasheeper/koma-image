let cur = 'small';
function render(k){
  cur = k; const s = S[k];
  document.querySelectorAll('#demo button').forEach(b => b.classList.toggle('on', b.dataset.k === k));
  $('#stream').innerHTML = `<div class="q">${s.q}</div><div class="trace">${s.trace}</div><div class="a">${s.a}</div>`;
  const c = $('#card'); c.className = 'card';
  c.innerHTML = `
    <div class="head">${ico(I[s.icon])}
      <div class="txt"><b>${s.title}</b><span>${s.desc}</span></div>
      <button class="act" id="ign" title="Esc">忽略</button>
      <button class="act pri" id="app" title="Tab">应用</button>
    </div>
    <div class="steps${s.steps.length > 1 ? ' multi' : ''}">${s.steps.map(step).join('')}</div>`;
  bind();
}

function bind(){
  const all = [...document.querySelectorAll('.step')];
  const free = all.filter(s => !s.hasAttribute('data-lock'));
  free.forEach(s => { const x = s.querySelector('.x');
    x.onclick = () => {
      s.classList.toggle('off');
      x.textContent = s.classList.contains('off') ? '恢复' : '撤回';
      const anyOn = all.some(n => n.hasAttribute('data-lock')) ? free.some(n => !n.classList.contains('off')) || true : free.some(n => !n.classList.contains('off'));
      $('#app').setAttribute('aria-disabled', !free.some(n => !n.classList.contains('off')) && !all.some(n => n.hasAttribute('data-lock')));
    };
  });
  document.querySelectorAll('.opt').forEach(o => o.onclick = () => {
    document.querySelectorAll('.opt').forEach(b => b.classList.toggle('on', b === o));
    $('#pn').hidden = o.dataset.v !== 'n'; $('#po').hidden = o.dataset.v !== 'o';
  });
  $('#app').onclick = apply; $('#ign').onclick = ignore;
}

function apply(){
  const a = $('#app'); if (!a || a.getAttribute('aria-disabled') === 'true') return;
  const s = S[cur];
  $('#card').innerHTML = `<div class="done">${ico(I.ok)}<b>${s.done}</b>· ${s.tail}<span class="sp"></span><button class="act" id="undo">撤销</button></div>`;
  $('#undo').onclick = () => { render(cur); $('#live').textContent = '已撤销'; };
  $('#live').textContent = `${s.done}，${s.tail}`;
}
function ignore(){
  if (!$('#app')) return;
  $('#card').classList.add('gone'); $('#live').textContent = '草稿已忽略';
}

$('#demo').onclick = e => e.target.dataset.k && render(e.target.dataset.k);
document.addEventListener('keydown', e => {
  if (e.key === 'Tab') { e.preventDefault(); apply(); }
  if (e.key === 'Escape') ignore();
});
render('small');
